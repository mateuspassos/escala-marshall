    <script>
      const INICIO = new Date('2026-11-09');
      const TURNOS = { inicio: '06:00', fim: '18:00' };
      const CICLO = [
        { tipo: 'trabalho', dias: 5 },
        { tipo: 'folga', dias: 2 },
        { tipo: 'trabalho', dias: 2 },
        { tipo: 'folga', dias: 5 }
      ];
      const DIAS_SEMANA = ['Dom','Seg','Ter','Qua','Qui','Sex','Sáb'];
      const somaDias = (d, n) => { const x=new Date(d); x.setDate(x.getDate()+n); return x; };
      const mesmoDia = (a,b) => a.toDateString()===b.toDateString();
      const formata = d => d.toLocaleDateString('pt-BR',{day:'2-digit',month:'2-digit'});
      function getStatus(d) {
        const diff=Math.floor((d-INICIO)/86400000);
        if(diff<0) return {tipo:'futuro',diaNoCiclo:-1};
        const len=CICLO.reduce((s,c)=>s+c.dias,0);
        const pos=diff%len;
        let acc=0;
        for(const b of CICLO){ if(pos<acc+b.dias) return {tipo:b.tipo,diaNoCiclo:pos,bloco:b};
  acc+=b.dias; }
        return {tipo:'trabalho',diaNoCiclo:pos};
      }
      function render(){
        const hoje=new Date(); hoje.setHours(0,0,0,0);
        let t=0,f=0;
        for(let i=0;i<30;i++){ const st=getStatus(somaDias(hoje,i)); st.tipo==='trabalho'?t++:f++; }
        document.getElementById('resumo').innerHTML=`<div class="card trabalho"><div
  class="label">Dias de Trabalho (30d)</div><div class="valor">${t}</div></div><div class="card
  folga"><div class="label">Dias de Folga (30d)</div><div class="valor">${f}</div></div>`;
        document.getElementById('cal-head').innerHTML=DIAS_SEMANA.map(d=>`<div>${d}</div>`).join('');
        const m=hoje.getMonth(), a=hoje.getFullYear();
        const first=new Date(a,m,1), last=new Date(a,m+1,0);
        let html=''.padStart(first.getDay(),'<div class="dia"></div>');
        for(let dia=1; dia<=last.getDate(); dia++){
          const data=new Date(a,m,dia), st=getStatus(data);
          const
  classes=['dia',st.tipo,mesmoDia(data,hoje)?'hoje':'',data<hoje?'passado':''].filter(Boolean).join('
  ');
          html+=`<div class="${classes}" title="${formata(data)} -
  ${st.tipo==='trabalho'?'Trabalho':'Folga'}"><span class="num">${dia}</span><span
  class="turno">${st.tipo==='trabalho'?TURNOS.inicio+'–'+TURNOS.fim:'—'}</span></div>`;
        }
        document.getElementById('cal-grid').innerHTML=html;
        const stHoje=getStatus(hoje);
        let prox=null;
        for(let i=1;i<=14;i++){ const d=somaDias(hoje,i);
  if(getStatus(d).tipo!==stHoje.tipo){prox={data:d,tipo:getStatus(d).tipo};break;} }
        document.getElementById('ciclo-info').innerHTML=`Hoje: <strong>${stHoje.tipo==='trabalho'?'🟦
  Trabalho':'🟩 Folga'}</strong>${prox?' | Próxima mudança: <strong>'+formata(prox.data)+'</strong>
  ('+(prox.tipo==='trabalho'?'Trabalho':'Folga')+')':''} | Ciclo: dia ${stHoje.diaNoCiclo+1} de 14`;
      }
      render();
    </script>
